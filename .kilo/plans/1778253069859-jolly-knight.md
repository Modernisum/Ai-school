#### T1.4 — Add Backend Token Expiry Check
**File:** `Backend/src/routes/ws.rs:102-120`

```rust
async fn authenticate_socket(
    socket: &mut WebSocket,
    state: &AppState,
) -> Result<(String, String, Option<String>), ()> {
    if let Some(Ok(msg)) = socket.recv().await {
        if let Message::Text(text) = msg {
            if let Ok(payload) = serde_json::from_str::<WsAuthPayload>(&text) {
                if let Ok(Some(token_data)) = state.repos.auth.get_token(&payload.token).await {
                    // ✅ ADD: Check expiry
                    let expires_at = token_data["expiresAt"]
                        .as_str()
                        .and_then(|s| chrono::DateTime::parse_from_rfc3339(s).ok());
                    if let Some(expiry) = expires_at {
                        if chrono::Utc::now() > expiry.with_timezone(&chrono::Utc) {
                            tracing::warn!("WS auth failed: token expired");
                            return Err(());
                        }
                    }
                    // ✅ END
                    let u_id = token_data["tokenId"]
                        .as_str()
                        .unwrap_or("unknown")
                        .to_string(); 
                    return Ok((u_id, payload.school_id, payload.vehicle_id));
                }
            }
        }
    }
    Err(())
}
```

#### T2.4 — Connection Pool & Rate Limiting
**File:** `Backend/src/routes/ws.rs`

```rust
use std::sync::atomic::{AtomicUsize, Ordering};
use std::collections::HashMap;
use std::sync::Mutex;
use tokio::sync::Semaphore;
use std::sync::Arc;

// Global connection limits
static ACTIVE_CONNECTIONS: AtomicUsize = AtomicUsize::new(0);
const MAX_CONNECTIONS: usize = 5000;
const MAX_PER_USER: usize = 3;

// User connection tracker
lazy_static! {
    static ref USER_CONNECTIONS: Mutex<HashMap<String, usize>> = Mutex::new(HashMap::new());
}

pub async fn ws_handler(ws: WebSocketUpgrade, State(state): State<AppState>) -> impl IntoResponse {
    // Rate limit check
    if ACTIVE_CONNECTIONS.load(Ordering::SeqCst) >= MAX_CONNECTIONS {
        return (StatusCode::SERVICE_UNAVAILABLE, "Server at capacity").into_response();
    }
    
    // Per-user concurrency limit
    // (We check inside handle_socket after auth)
    
    ws.on_upgrade(move |socket| {
        ACTIVE_CONNECTIONS.fetch_add(1, Ordering::SeqCst);
        handle_socket(socket, state).await;
        ACTIVE_CONNECTIONS.fetch_sub(1, Ordering::SeqCst);
    })
}
```
